class BankAccount {
  constructor(){
    this.balance = 0
    this.transactions = []
  }

  deposit(amount){
    if(typeof amount !== "number") {
        alert("Please enter a number")
        return;
        }
    if(amount > 0){
      this.transactions.push({
        type: "deposit",
        amount: amount
      }) 
      this.balance += amount;
      return `Successfully deposited $${amount}. New balance: $${this.balance}`
      }
    if(amount < 0 || amount === 0){
        return "Deposit amount must be greater than zero."
      }
    }

  withdraw(amount){
    if(typeof amount !== "number") {
        alert("Please enter a number")
        return;
        }
    if(amount < 0 || amount === 0 || amount > this.balance) {
        return "Insufficient balance or invalid amount."
      }
    if(amount > 0){
      this.transactions.push({
        type: "withdraw",
        amount: amount
      }) 
      this.balance -= amount;
      return `Successfully withdrew $${amount}. New balance: $${this.balance}`
      }
    }
  checkBalance(){
    return `Current balance: $${this.balance}`
  }
  listAllDeposits(){
    let returnMsg = "Deposits: "
    const deposits = this.transactions.filter((transaction) => transaction.type === "deposit")
    deposits.forEach((transaction) => returnMsg += `${transaction.amount},`)
    return returnMsg.slice(0, -1)
  }

  listAllWithdrawals(){
    let returnMsg = "Withdrawals: "
    const withdrawals = this.transactions.filter((transaction) => transaction.type === "withdraw")
    withdrawals.forEach((transaction) => returnMsg += `${transaction.amount},`)
    return returnMsg.slice(0, -1)
  }
  }

  const myAccount = new BankAccount
  myAccount.deposit(20)
  myAccount.deposit(31)
  myAccount.deposit(64)
  myAccount.deposit(11)
  myAccount.withdraw(1000)
  myAccount.withdraw(10)
  myAccount.withdraw(5)
  myAccount.withdraw(300)
  myAccount.withdraw(12)
  myAccount.deposit(45)
  myAccount.deposit(64)
  myAccount.deposit(100)
