const { useState, useMemo } = React;

function handleConvert(amount, convertFrom) {
  const valueMapping = {
    USD: 1,
    EUR: 0.92,
    GBP: 0.78,
    JPY: 156.7
};



const calculatedValue = {
  USD: valueMapping["USD"] / valueMapping[convertFrom] * amount,
  EUR: valueMapping["EUR"] / valueMapping[convertFrom] * amount,
  GBP: valueMapping["GBP"] / valueMapping[convertFrom] * amount,
  JPY: valueMapping["JPY"] / valueMapping[convertFrom] * amount,
}

console.log("Im recalculating")

return calculatedValue;



  // const calculatedConversion = valueMapping[convertTo] / valueMapping[convertFrom] * amount;

  // console.log(calculatedConversion)

  // return calculatedConversion.toFixed(2);

}



export function CurrencyConverter() {
  const [amount, setAmount] = useState(0);
  const [convertFrom, setConvertFrom] = useState("USD")
  const [convertTo, setConvertTo] = useState("JPY")
  const convertedCurrency = useMemo(() => handleConvert(amount, convertFrom), [amount, convertFrom])



  return (
    <>
    <header>
    <h1>Currency Converter</h1>
    </header>
    <form>
      <label htmlFor="amount">Amount: </label>
      <input id="amount" value={amount} onChange={((e) => setAmount(e.target.value))} type="number" />
      <div className="select-container">
      <div className="select-wrapper">
      <label htmlFor="convert-from">Convert from: </label>
        <select value={convertFrom} onChange={(e) => setConvertFrom(e.target.value)}>
          <option>USD</option>
          <option>EUR</option>
          <option>GBP</option>
          <option>JPY</option>
        </select>
        </div>
        <div className="select-wrapper">
          <label htmlFor="convert-to">Convert to:</label>
          <select value={convertTo} id="convert-to" onChange={(e) => setConvertTo(e.target.value)}>
            <option>USD</option>
            <option>EUR</option>
            <option>GBP</option>
            <option>JPY</option>
          </select>
        </div>
      </div>
      </form>
      <div className="results-container">
        <p>Converting from {convertFrom} to {convertedCurrency[convertTo].toFixed(2)} {convertTo}</p>
      </div>
    </>
  )
}
