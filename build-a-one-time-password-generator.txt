** start of index.jsx **

const { useState, useEffect, useRef } = React;

export const OTPGenerator = () => {

  const [countdown, setCountdown] = useState(null);
  const [isActive, setIsActive] = useState(false)
  const [otp, setOtp] = useState(null);;
  useEffect(() => {
    const handler = setTimeout(() => {
      if(countdown > 0) {
        setCountdown(countdown - 1)
      }
    }, 1000)
      

    return () => {
      clearTimeout(handler)
      };
  },[countdown])

function getRandomInt(min, max) {
  const minCeiled = Math.ceil(min);
  const maxFloored = Math.floor(max);
  return Math.floor(Math.random() * (maxFloored - minCeiled) + minCeiled); // 
}

  const handleOnclick = () => {
    setCountdown(5)
    setOtp(getRandomInt(100000,999999))
  }

  return(
    <div className="container">
      <h1 id="otp-title">OTP Generator</h1>
      <h2 id="otp-display">
        {countdown === null && "Click 'Generate OTP' to get a code"}
        {countdown >= 0 && otp}
        </h2>
      {countdown  > 0 && <p id="otp-timer">Expires in: {countdown} seconds</p>}
      {countdown  === 0 && <p id="otp-timer">OTP expired. Click the button to generate a new OTP.</p>}
      <button disabled={countdown > 0} onClick={handleOnclick} id="generate-otp-button">Generate OTP</button>
    </div>
  )

};
